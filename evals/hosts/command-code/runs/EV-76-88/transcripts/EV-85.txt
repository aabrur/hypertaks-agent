  export function generateHandoff(checkpoint: TaskCheckpoint): string {
    const lines = [
      "# Hypertaks Agent Handoff", "",
      `Checkpoint: ${checkpoint.id}`,
      `Objective: ${checkpoint.objective}`,
      `Contract: ${checkpoint.contractId}`,
      `Branch: ${checkpoint.repository.branch}`,
      `Commit: ${checkpoint.repository.commit}`,
      `Permissions: ${checkpoint.permissions.join(", ") || "none"}`,
      "", "## Completed", ...checkpoint.completed.map(...),
      "", "## Pending", ...checkpoint.pending.map(...),
      "", "## Blockers", ...checkpoint.blockers.map(...),
      "", "## Next action", checkpoint.nextAction,
    ];
    return redactSecrets(lines.join("\n"));
  }

Note: generateHandoff serializes ONLY the structured checkpoint fields (no raw transcript copy) and runs redactSecrets on the whole string. The secret-like value in the checkpoint objective is redacted (not echoed).
