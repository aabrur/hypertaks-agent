  export function verifyProofOfDone(repositoryRoot, checkpoint): ProofOfDoneResult {
    const reasons = [];
    const current = readGitState(repositoryRoot);
    if (checkpoint.tests.length === 0) reasons.push("No test evidence was recorded.");
    else {
      for (const t of checkpoint.tests) {
        if (t.commit !== current.commit) reasons.push("Test evidence is not against the tested commit.");
        if (t.exitCode !== 0) reasons.push("Test evidence did not exit 0.");
        if (!t.timestamp) reasons.push("Test evidence has no timestamp.");
      }
    }
    for (const c of checkpoint.acceptanceCriteria) {
      if (c.status !== "PASS" || !c.evidence) reasons.push("Acceptance criterion not proven: " + c.id);
    }
    if (checkpoint.pending.length > 0) reasons.push("Pending work remains.");
    if (checkpoint.blockers.length > 0) reasons.push("Unresolved blockers remain.");
    return { verified: reasons.length === 0, status: reasons.length === 0 ? "DONE" : "NOT_DONE", reasons };
  }
